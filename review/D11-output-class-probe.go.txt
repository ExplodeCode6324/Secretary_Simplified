package tests

import (
	"bytes"
	"context"
	"encoding/json"
	ctxbuild "secretarysimplified/context"
	"secretarysimplified/contract"
	"secretarysimplified/core"
	"secretarysimplified/model"
	"testing"
	"time"
)

type outputClassProbe struct {
	model.Client
	seen string
}

func (p *outputClassProbe) Generate(ctx context.Context, r model.Request) (model.Result, error) {
	p.seen = r.DataClass
	raw, e := model.Fixture(r)
	if e != nil {
		return model.Result{}, e
	}
	var d contract.DecisionEnvelope
	json.Unmarshal(raw, &d)
	d.Reply["questions"] = []any{map[string]any{"text": "UNIQUE_ASSISTANT_QUERY_D11 confirm SYNTHETIC_CANARY_PERSONAL_864?", "item_id": nil}}
	raw, e = json.Marshal(d)
	return model.Result{Output: raw}, e
}
func TestProbeOutputClassInheritance(t *testing.T) {
	s, c, _ := setup(t)
	ctx := context.Background()
	session := contract.NewID()
	c.Policy.AllowedClasses = []string{"SYNTHETIC", "PERSONAL"}
	p := &outputClassProbe{Client: model.New(c)}
	prior := input(session)
	prior.DataClass = "PERSONAL"
	prior.Text = "SYNTHETIC_CANARY_PERSONAL_864"
	pt, e := s.AcceptInput(ctx, prior, 100)
	if e != nil {
		t.Fatal(e)
	}
	if e = s.FinishTurn(ctx, pt.ID, map[string]any{"text": "recorded", "evidence": []any{}}, []string{}, nil); e != nil {
		t.Fatal(e)
	}
	ask := input(session)
	ask.Text = "Ask a clarification about the prior value"
	at, e := s.AcceptInput(ctx, ask, 100)
	if e != nil {
		t.Fatal(e)
	}
	svc := core.Service{Store: s, Config: c, Model: p, GrantID: runtimeGrant(t, s)}
	if e = svc.Process(ctx, at); e != nil {
		t.Fatal(e)
	}
	if p.seen != "PERSONAL" {
		t.Fatal("precondition effective class", p.seen)
	}
	page, e := s.SearchMemoryPage(ctx, "UNIQUE_ASSISTANT_QUERY_D11", nil, nil)
	if e != nil {
		t.Fatal(e)
	}
	if len(page.Events) != 1 || page.Events[0].Role != "ASSISTANT" {
		t.Fatal("query not isolated", page)
	}
	c.Policy.AllowedClasses = []string{"SYNTHETIC"}
	next := input(contract.NewID())
	next.Text = "Retrieve the prior clarification"
	nt, e := s.AcceptInput(ctx, next, 100)
	if e != nil {
		t.Fatal(e)
	}
	b := ctxbuild.Builder{Store: s, Config: c, Model: model.New(c), RetrievalServed: true, RetrievedEvents: page.Events, RetrievedClasses: page.DataClasses}
	req, _, e := b.Build(ctx, nt, time.Now())
	if e != nil {
		t.Fatalf("probe blocked at builder: %v classes=%v", e, page.DataClasses)
	}
	wire, e := model.New(c).Encode(req)
	if e != nil {
		t.Fatal(e)
	}
	present := bytes.Contains(wire, []byte("SYNTHETIC_CANARY_PERSONAL_864"))
	t.Logf("creation_effective_class=%s creation_input_class=%s retrieved_classes=%v retrieved_role=%s new_request_class=%s synthetic_only_encode=PASS canary_in_wire=%t", p.seen, ask.DataClass, page.DataClasses, page.Events[0].Role, req.DataClass, present)
	if !present {
		t.Fatal("canary not in wire")
	}
}
