// Independent /tmp probe: drives the exported Model through the real code paths
// with a contract-shaped fake caller. Not part of the repository test suite.

type probeCaller struct {
	mu          sync.Mutex
	calls       []string
	bodies      []contract.InputEnvelope
	authority   Authority
	turnID      string
	submitTurn  string
	queryCode   string
	queryStatus int
	revision    float64
	known       map[string]bool
}

func probeRow(id string, rev float64) map[string]any {
	return map[string]any{"id": id, "title": id, "revision": rev, "enabled": true, "state": "RUNNING"}
}
func probeStr(s string) *string { return &s }

func (f *probeCaller) Call(ctx context.Context, method, path string, body any) (transport.Envelope, int, error) {
	f.mu.Lock()
	defer f.mu.Unlock()
	f.calls = append(f.calls, method+" "+path)
	switch {
	case method == "POST" && path == "/v1/inputs":
		in := body.(contract.InputEnvelope)
		f.bodies = append(f.bodies, in)
		return transport.Envelope{Result: map[string]any{"turn_id": f.submitTurn}}, 202, nil
	case path == "/v1/conversation":
		return transport.Envelope{Result: f.authority}, 200, nil
	case strings.HasPrefix(path, "/v1/conversation/history"):
		return transport.Envelope{Result: History{SessionID: f.authority.SessionID}}, 200, nil
	case strings.HasPrefix(path, "/v1/requests/"):
		rid := strings.TrimPrefix(path, "/v1/requests/")
		if f.known[rid] {
			return transport.Envelope{Result: map[string]any{"turn_id": f.turnID}}, 200, nil
		}
		return transport.Envelope{Error: map[string]any{"code": "NOT_FOUND"}}, 404, nil
	case strings.HasPrefix(path, "/v1/turns/"):
		if f.queryStatus != 0 {
			return transport.Envelope{Error: map[string]any{"code": f.queryCode}}, f.queryStatus, nil
		}
		rid := ""
		if len(f.bodies) > 0 {
			rid = f.bodies[0].RequestID
		}
		return transport.Envelope{Result: contract.InputTurn{ID: f.turnID, RequestID: rid, State: "ACCEPTED", SessionID: f.authority.SessionID}}, 200, nil
	case path == "/v1/tasks?limit=50":
		return transport.Envelope{Result: Page{Items: []map[string]any{probeRow("t1", f.revision), probeRow("t2", f.revision)}, NextCursor: probeStr("p2")}}, 200, nil
	case path == "/v1/tasks?limit=50&cursor=p2":
		return transport.Envelope{Result: Page{Items: []map[string]any{probeRow("t2", f.revision), probeRow("t3", f.revision)}, NextCursor: probeStr("p3")}}, 200, nil
	}
	return transport.Envelope{Result: map[string]any{}}, 200, nil
}
func probeSetup(t *testing.T) (Model, *probeCaller) {
	t.Helper()
	m := testModel(t)
	f := &probeCaller{turnID: contract.NewID(), authority: m.authority, revision: 1, known: map[string]bool{}}
	f.submitTurn = f.turnID
	m.opts.Core = f
	m.opts.Runner = f
	return m, f
}

// P2: after a legal 202 acceptance, an observation GET whose code is itself on
// the precommit whitelist must never restore the draft or drop tracking.
func TestProbeAcceptedObservationWhitelistedCodes(t *testing.T) {
	for _, c := range []struct {
		status int
		code   string
	}{
		{401, "UNAUTHENTICATED"},
		{403, "QUESTION_AUTHORITY_DENIED"},
		{404, "QUESTION_NOT_FOUND_IN_SESSION"},
		{400, "BACKPRESSURE"},
	} {
		t.Run(c.code, func(t *testing.T) {
			m, f := probeSetup(t)
			f.queryStatus, f.queryCode = c.status, c.code
			m.input.SetValue("原始合成输入")
			m, _ = step(m, m.send()())
			id := f.bodies[0].RequestID
			if len(f.bodies) != 1 || len(f.calls) != 2 || f.calls[1] != "GET /v1/turns/"+f.turnID {
				t.Fatal("unexpected calls", f.calls)
			}
			if _, ok := m.accepted[id]; !ok {
				t.Fatal("accepted fact lost", m.accepted)
			}
			if m.turns[f.turnID].ID != f.turnID || m.turns[f.turnID].RequestID != id {
				t.Fatal("known turn identity lost", m.turns)
			}
			if m.pending[id] == "" || !residualPendingFile(m, id) {
				t.Fatal("tracking lost")
			}
			if m.input.Value() != "" {
				t.Fatal("draft restored", m.input.Value())
			}
			if strings.Contains(m.notice, "被后端拒绝") || !strings.Contains(m.notice, "已受理") {
				t.Fatal("notice", m.notice)
			}
		})
	}
}

// P3: a 202 without a decodable turn_id keeps submit-unknown, never restores the
// draft, never re-POSTs, and recovers the accepted identity from the receipt.
func TestProbeUnconfirmed202RecoversViaReceipt(t *testing.T) {
	m, f := probeSetup(t)
	f.submitTurn = ""
	m.input.SetValue("未知受理形状")
	m, _ = step(m, m.send()())
	id := f.bodies[0].RequestID
	if len(f.calls) != 1 {
		t.Fatal("unexpected calls", f.calls)
	}
	if m.input.Value() != "" {
		t.Fatal("draft restored")
	}
	if _, ok := m.accepted[id]; ok {
		t.Fatal("accepted without legal receipt")
	}
	if m.pending[id] == "" {
		t.Fatal("tracking lost")
	}
	f.known = map[string]bool{id: true}
	m.syncing = false
	m, _ = step(m, m.sync()())
	if m.accepted[id] != f.turnID {
		t.Fatal("accepted not recovered from receipt", m.accepted)
	}
	if m.turns[f.turnID].RequestID != id {
		t.Fatal("identity not rebound")
	}
	if len(f.bodies) != 1 {
		t.Fatal("submission retried")
	}
}

// P1: an ID duplicated across loaded pages must show the refreshed copy after the
// anchor-page refresh (observable revision is the one the confirm dialog binds).
func TestProbePanelDuplicateAnchorFreshness(t *testing.T) {
	m, f := probeSetup(t)
	cmd := m.showPanel("tasks")
	m, _ = step(m, cmd())
	m, cmd = step(m, tea.KeyMsg{Type: tea.KeyRunes, Runes: []rune("n")})
	m, _ = step(m, cmd())
	if len(m.rows) != 3 {
		t.Fatal("precondition rows", len(m.rows))
	}
	m.selected = 1
	if id, _ := m.rows[1]["id"].(string); id != "t2" {
		t.Fatal("precondition selection")
	}
	f.revision = 2
	m, _ = step(m, m.openPanel("tasks", false)())
	if id, _ := m.rows[m.selected]["id"].(string); id != "t2" {
		t.Fatal("selection moved", m.rows[m.selected])
	}
	if rev, _ := m.rows[m.selected]["revision"].(float64); rev != 2 {
		t.Fatal("refreshed copy hidden by duplicate", m.rows[m.selected])
	}
	if len(m.rows) != 3 {
		t.Fatal("range replaced", len(m.rows))
	}
	if m.panelCursor == nil || *m.panelCursor != "p3" {
		t.Fatal("append anchor changed", m.panelCursor)
	}
}
