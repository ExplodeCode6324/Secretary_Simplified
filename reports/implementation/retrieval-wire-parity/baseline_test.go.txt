package baseline

import (
	"bytes"
	"context"
	"path/filepath"
	"secretarysimplified/config"
	current "secretarysimplified/context"
	"secretarysimplified/contract"
	"secretarysimplified/model"
	"secretarysimplified/store"
	"testing"
	"time"
)

func TestEmptyRetrievalWireIdenticalToGitBaseline(t *testing.T) {
	d := t.TempDir()
	s, e := store.Init(filepath.Join(d, "db"), filepath.Join(d, "objects"))
	if e != nil {
		t.Fatal(e)
	}
	defer s.Close()
	ctx := context.Background()
	c := config.Default(d)
	p := model.New(c)
	now := time.Now().UTC()
	id := contract.NewID()
	it := contract.Item{SchemaVersion: 1, ID: id, Revision: 1, Domain: "work", Kind: "TASK", Title: "baseline item", Status: "OPEN", Priority: 1, Timezone: "UTC", TimeState: "UNKNOWN", DependencyIDs: []string{}, Evidence: []contract.EvidenceRef{}, CreatedAt: contract.Timestamp(now), UpdatedAt: contract.Timestamp(now), Extensions: map[string]any{}}
	if e = s.PutItem(ctx, it, 0); e != nil {
		t.Fatal(e)
	}
	for _, text := range []string{"simple request", "update item " + id} {
		in := contract.InputEnvelope{SchemaVersion: 1, RequestID: contract.NewID(), SessionID: contract.NewID(), PrincipalID: "master", Origin: "MASTER_CLI", ReceivedAt: contract.Timestamp(now), Text: text, AttachmentRefs: []contract.ObjectRef{}, DataClass: "SYNTHETIC", Extensions: map[string]any{}}
		turn, e := s.AcceptInput(ctx, in, 100)
		if e != nil {
			t.Fatal(e)
		}
		old := Builder{Store: s, Model: p, Config: c}
		new := current.Builder{Store: s, Model: p, Config: c}
		a, _, e := old.Build(ctx, turn, now)
		if e != nil {
			t.Fatal(e)
		}
		b, _, e := new.Build(ctx, turn, now)
		if e != nil {
			t.Fatal(e)
		}
		shared := contract.NewID()
		a.ContextID = shared
		b.ContextID = shared
		a.Input.(map[string]any)["context_id"] = shared
		b.Input.(map[string]any)["context_id"] = shared
		ab, e := p.Encode(a)
		if e != nil {
			t.Fatal(e)
		}
		bb, e := p.Encode(b)
		if e != nil {
			t.Fatal(e)
		}
		if !bytes.Equal(ab, bb) {
			t.Fatal("wire differs without retrieval")
		}
		t.Logf("PASS no-retrieval wire parity: %d bytes", len(ab))
	}
}
